cmake_minimum_required(VERSION 3.8)
project(mrider_navigation_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(Boost REQUIRED COMPONENTS thread serialization)
find_package(TBB REQUIRED)
add_executable(managed_slam_node src/managed_slam_node.cpp)
target_compile_features(managed_slam_node PRIVATE cxx_std_17)
ament_target_dependencies(managed_slam_node rclcpp slam_toolbox)
# Humble does not export imported targets for these two component libraries.
# Avoid linking its entire transitive export list (including RViz plugins).
find_library(MRIDER_SLAM_ASYNC_LIBRARY async_slam_toolbox HINTS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
find_library(MRIDER_SLAM_COMMON_LIBRARY toolbox_common HINTS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
find_library(MRIDER_KARTO_LIBRARY kartoSlamToolbox HINTS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
target_link_libraries(managed_slam_node ${MRIDER_SLAM_ASYNC_LIBRARY} ${MRIDER_SLAM_COMMON_LIBRARY} ${MRIDER_KARTO_LIBRARY} Boost::thread Boost::serialization TBB::tbb)
install(TARGETS managed_slam_node DESTINATION lib/${PROJECT_NAME})
add_executable(managed_controller_node src/managed_controller_node.cpp)
target_compile_features(managed_controller_node PRIVATE cxx_std_17)
ament_target_dependencies(managed_controller_node rclcpp nav2_controller)
install(TARGETS managed_controller_node DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/rviz_supervisor scripts/laser_odometry_node scripts/observed_map_node scripts/camera_rules_node scripts/world_node scripts/safety_node scripts/mission_node scripts/run_scenario DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz licenses DESTINATION share/${PROJECT_NAME})
